\begin{frame}[fragile]{Week 8 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.3em]
      \small
      \item[\textbf{1}] \textbf{Is-a.} \code{class Dog extends Animal}: a \code{Dog}
            \emph{is an} \code{Animal} and inherits its fields and methods. Python:
            \code{class Dog(Animal):}.
      \item[\textbf{2}] \textbf{\code{super}.} A constructor first runs
            \code{super(...)} to build the parent part (Python:
            \code{super().\_\_init\_\_(...)}). \code{super.m()} runs the
            \textbf{parent's} \code{m}.
      \item[\textbf{3}] \textbf{Override vs overload.} Same name \textbf{and}
            parameters: it \emph{overrides} (\code{@Override} makes the compiler
            check). Different parameters: it \emph{overloads}, a separate method.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.3em]
      \small
      \item[\textbf{4}] \textbf{Declared type vs object.} \code{Animal a = new Dog();}
            The declared type decides what you may \textbf{call}; the object decides
            which override \textbf{runs}.
      \item[\textbf{5}] \textbf{Promises.} An \code{abstract} class cannot be
            \code{new}ed. An \code{interface} lists public methods; a class may
            \code{implement} many. Python: \code{ABC}, or \emph{duck typing}.
    \end{itemize}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
interface Priced { int cost(); }
abstract class Item implements Priced {
    final String name;
    Item(String n) { name = n; }
}
class Book extends Item {
    Book(String n) { super(n); }
    @Override public int cost() { return 20; }
}
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{2}{Multiple choice}Which line does not compile?}
  \begin{columns}[T,onlytextwidth]

\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
interface Speaker { String speak(); }
abstract class Pet implements Speaker {
    protected final String name;
    Pet(String name) { this.name = name; }

class Cat extends Pet {
    Cat(String name) { super(name); }

\end{lstlisting}
    \begin{discussionbox}{Choose one (60 seconds)}
      \small
      \wrongopt{A}{\code{Speaker s = p;}}\\[0.15em]
      \wrongopt{B}{\code{String t = p.speak();}}\\[0.15em]
      \rightopt{C}{\code{p.purr();}}\\[0.15em]
      \wrongopt{D}{\code{new Cat("Mo").purr();}}
    \end{discussionbox}
    \column{0.46\textwidth}
    {\small \code{main} is in another class, \code{Demo}, in the same file, and
    starts with:}
\begin{lstlisting}[language=Java]
Pet p = new Cat("Tom");
\end{lstlisting}
    \begin{onlyenv}<1>
    {\small Each option is the next line of \code{main}. Exactly \textbf{one} of them
    does not compile.}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answer: C}
      \footnotesize\raggedright
      The object is a \code{Cat}, but the compiler checks the \textbf{declared}
      type, \code{Pet}, which has no \code{purr}:
      {\scriptsize\code{cannot find symbol}}\\[0.2em]
      \textbf{A}: \code{Pet implements Speaker}: a \code{Pet} \emph{is a}
      \code{Speaker}.\\[0.1em]
      \textbf{B}: \code{Pet} promises \code{speak()} through \code{Speaker}.\\[0.1em]
      \textbf{D}: here the declared type is \code{Cat}.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{3}{Multiple choice}Overloading or overriding?}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
class Animal {
    String sound() { return "..."; }
}
class Dog extends Animal {
    @Override String sound() { return "woof"; }
}
class Vet {
    static String check(Animal a) {
        return "animal " + a.sound();
    }
    static String check(Dog d) {
        return "dog " + d.sound();
    }
}
// in main, in another class Demo
Animal pet = new Dog();
System.out.println(Vet.check(pet));
\end{lstlisting}
    \column{0.46\textwidth}
    \begin{discussionbox}{What is printed? (90 seconds)}
      \small
      \wrongopt{A}{\code{dog woof}}\\[0.05em]
      \rightopt{B}{\code{animal woof}}\\[0.05em]
      \wrongopt{C}{\code{animal ...}}\\[0.05em]
      \wrongopt{D}{\code{dog ...}}
    \end{discussionbox}
    \begin{onlyenv}<1>
    {\small \neutrallabel{Hint:} two different choices happen here. Which
    \code{check}? Then, inside it, which \code{sound}?}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answer: B}
      \footnotesize\raggedright
      \textbf{Which \code{check}} (overload): the compiler picks it from
      \code{pet}'s declared type, \code{Animal}.\\[0.1em]
      \textbf{Which \code{sound}} (override): picked as it runs, from the object, a
      \code{Dog}.\\[0.1em]
      \textbf{A}, \textbf{D}: an overload does not follow the object.
      \textbf{C}: an override does.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}Cinema tickets (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
interface Priced { int price(); }
public class Ticket implements Priced {
    protected final String film;
    Ticket(String film) { this.film = film; }
    int price() { return 15; }
    String label() {
        return film + " $" + price();
    }
}
class Student extends Ticket {
    Student(String film) { super("film"); }
    public int price() { return price() - 5; }
    String label(int seat) {
        return "Student " + super.label();
    }
}
\end{lstlisting}
    {\small \badlabel{Find 4 bugs} (3 min): 1 compile error, 3 logic errors. Give each
    line number and its fix.}
    \column{0.44\textwidth}
    \begin{onlyenv}<1>
    \neutrallabel{What the classes should do}
    \begin{itemize}[itemsep=0.05em, topsep=0.1em, leftmargin=1em]
      \footnotesize
      \item A ticket costs 15; a student ticket 5 \textbf{less}
      \item \code{Student}'s \code{label()} starts \code{"Student "}
    \end{itemize}
    \neutrallabel{Test} {\footnotesize (\code{main} in class \code{Demo})}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
Ticket a = new Ticket("Dune");
Ticket b = new Student("Emma");
System.out.println(a.label());
System.out.println(b.label());
Priced p = b;
System.out.println(p.price());
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
Dune $15
Student Emma $10
10
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.2em, leftmargin=1em]
      \footnotesize
      \item \textbf{Line 5} (compile error): interface methods are \code{public},
            and an implementation may not be less visible:
            \code{attempting to assign weaker access privileges; was public}. Add
            \code{public}.
      \item \textbf{Line 11}: passes the text \code{"film"}, not the parameter. Use
            \code{super(film)}.
      \item \textbf{Line 12}: \code{price()} calls this same method again, forever
            (\code{StackOverflowError}). Use \code{super.price() - 5}.
      \item \textbf{Line 13}: different parameters, so it \emph{overloads}:
            \code{b.label()} still runs \code{Ticket}'s. Write
            \code{@Override String label()}.
    \end{itemize}
    {\footnotesize Fix only line 5: \code{Dune \$15}, then a crash.}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{7}{Find the bug}Staff pay (Python)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Py,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self.salary = salary
    def pay(self):
        return self.salary
    def __str__(self):
        return self.name + ": " + str(self.pay())

class Manager(Employee):

for (String s : lines) {
    try {
        axles += Integer.parseInt(s);
    } catch (NumberFormatException e) {
        System.out.println("skip " + s);
    }
}
\end{lstlisting}
