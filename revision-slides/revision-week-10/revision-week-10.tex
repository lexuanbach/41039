\begin{frame}[fragile]{Week 10 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{1}] \textbf{Output is a contract.} Fix the field order, the
            delimiter and the line ending. \code{println} ends the line;
            \code{print} and Python's \code{write} add \textbf{nothing}.
      \item[\textbf{2}] \textbf{Overwrite or append.} \code{"w"} and
            \code{new PrintWriter(name)} \textbf{empty} an existing file.
            \code{"a"} and \code{new FileWriter(name, true)} add to the end.
      \item[\textbf{3}] \textbf{Close it.} Output waits in a buffer until it is
            flushed. \code{try}-with-resources and \code{with} close (and flush) the
            file, even on an exception. \code{PrintWriter}'s \code{println} never
            throws: ask \code{checkError()}.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{4}] \textbf{Delimiters need a rule.} A comma \emph{inside} a name
            breaks \code{split(",")}. Python's \code{csv} module quotes such a field
            and reads it back whole. Everything read back is \textbf{text}.
      \item[\textbf{5}] \textbf{Calculate, format, write: three jobs.} A method that
            \emph{returns} the text can be tested in memory (\code{StringWriter},
            \code{StringIO}) without a real file.
    \end{itemize}
\begin{lstlisting}[language=Java]
try (PrintWriter out = new PrintWriter(
         new FileWriter("r.csv", true))) {
    out.println("Linh,85");  // ends the line
}  // closed and flushed here
\end{lstlisting}
\begin{lstlisting}[language=Py]
with open("r.csv", "a") as f:
    f.write("Linh,85\n")  # you add \n
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{3}{Multiple choice}\code{with} and an exception}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
\begin{lstlisting}[language=Py]
try:
    with open("r.txt", "w") as f:
        f.write("a\n")
        n = int("two")
        f.write("b\n")
except ValueError:
    print("closed?", f.closed)
with open("r.txt") as f:
    print(repr(f.read()))
\end{lstlisting}
    \begin{discussionbox}{What does it print? (60 seconds)}
      \small
      \wrongopt{A}{\makebox[7.9em][l]{\code{closed? False}}then \code{''}}\\[0.15em]
      \rightopt{B}{\makebox[7.9em][l]{\code{closed? True}}then \code{'a\textbackslash n'}}\\[0.15em]
      \wrongopt{C}{\makebox[7.9em][l]{\code{closed? True}}then \code{'a\textbackslash nb\textbackslash n'}}\\[0.15em]
      \wrongopt{D}{\makebox[7.9em][l]{\code{closed? False}}then \code{'a\textbackslash n'}}
    \end{discussionbox}
    \column{0.46\textwidth}
    {\small \code{int("two")} raises a \code{ValueError}. \code{repr} shows the text
    with quotes, and a newline as \code{\textbackslash n}.}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answer: B}
      \footnotesize\raggedright
      \code{with} closes the file when the block ends for \textbf{any} reason,
      including an exception. So \code{f.closed} is \code{True}: \textbf{A} and
      \textbf{D} are wrong.\\[0.2em]
      Closing also flushes, so the \code{"a\textbackslash n"} already written is
      saved: not \code{''} (\textbf{A}).\\[0.2em]
      The exception jumps straight to \code{except}, so \code{f.write("b\textbackslash n")}
      never runs (\textbf{C}).\\[0.2em]
      Java's \code{try}-with-resources does the same.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}The report file (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
import java.io.*;
public class Report {
    // header, a line per student, pass count
    static void save(String f, String[] names,
                     int[] marks) {
        FileWriter fw = new FileWriter(f);
        PrintWriter out = new PrintWriter(fw);
        out.println("name,mark");
        int passed = 0;
        for (int i = 0; i < names.length; i++) {
            out.print(names[i] + "," + marks[i]);
            if (marks[i] > 50) passed++;
        }
        out.println("passed," + passed);
    }
}
\end{lstlisting}
    {\small \badlabel{Find 4 bugs} (3 min): 1 compile error, 3 logic errors. Give each
    line number and its fix.}
    \column{0.44\textwidth}
    \begin{onlyenv}<1>
    \neutrallabel{What the method should do}
    \begin{itemize}[itemsep=0.05em, topsep=0.1em, leftmargin=1em]
      \footnotesize
      \item Write \code{name,mark}, one \code{name,mark} line per student, then
            \code{passed,<count>}
      \item A mark of 50 or more is a pass
      \item When \code{save} returns, every line is in the file
    \end{itemize}
    \neutrallabel{Test}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
// main (throws IOException) is in class Demo
String[] names = {"An", "Binh", "Chi"};
int[] marks = {50, 49, 90};
Report.save("r.csv", names, marks);
// then print every line of r.csv
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
name,mark
An,50
Binh,49
Chi,90
passed,2
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.25em, leftmargin=1em]
      \footnotesize
      \item \textbf{Line 6} (compile error): \code{new FileWriter} can throw an
            \code{IOException}, and nothing catches it. javac:
            {\scriptsize\code{unreported exception IOException; must be caught or
            declared to be thrown}}. Add \code{throws IOException} on line 5.
      \item \textbf{Line 11}: \code{print} ends no line, so all students run
            together. Use \code{println}.
      \item \textbf{Line 12}: \code{>} makes An's 50 a fail. Use \code{>=}.
      \item \textbf{Lines 6--7}: \code{out} is never closed, so the text stays in
            the buffer. Open it in
            \code{try (PrintWriter out = ...) \{ ... \}}.
    \end{itemize}
    {\footnotesize Fix only line 6 and the test prints \textbf{nothing}: \code{r.csv}
    is empty.}
    \end{onlyenv}
  \end{columns}

class Rows {
    static String row(String name, int mark) {
        String result = "fail";
        if (mark >= 50) result = "pass";
        return name + "," + mark + "," + result;
    }
    static String summary(int[] marks) {
        if (marks.length == 0) return "no marks";
        int passed = 0;
        for (int m : marks) if (m >= 50) passed++;
        return passed + "/" + marks.length + " passed";
    }
}
