\begin{frame}<1-2>{Fill the blanks: types and booleans}
  \begin{columns}[T,onlytextwidth]

    \begin{itemize}[itemsep=0.45em]
      \item Java is \blank{1} typed: every type is known at \blank{2}.
      \item \code{10 + 3} has type \blank{3}.\\ \code{i > 10} has type \blank{4}.
      \item \blank{5} assigns. \blank{6} compares.
      \item \code{a \&\& b} is true when \blank{7} sides are.\\
        \code{a || b} is true when \blank{8} side is.
    \end{itemize}
    \column{0.43\textwidth}
    \alt<1>{%
    \begin{discussionbox}{Word bank}
      \footnotesize
      \renewcommand{\arraystretch}{1.1}
      \begin{tabular}{@{}l@{\hspace{0.8em}}l@{}}
        \textbf{a)} dynamically    & \textbf{g)} \code{=}\\
        \textbf{b)} \code{boolean} & \textbf{h)} run time\\
        \textbf{c)} statically     & \textbf{i)} both\\
        \textbf{d)} \code{String}  & \textbf{j)} \code{==}\\
        \textbf{e)} compile time   & \textbf{k)} either\\
        \textbf{f)} \code{int}     & \textbf{l)} \code{!=}\\
      \end{tabular}
    \end{discussionbox}}{%
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{(1)} c) \textbf{statically} \quad \textbf{(2)} e) \textbf{compile time}\\[0.25em]
      \textbf{(3)} f) \code{int} \quad \textbf{(4)} b) \code{boolean} --- a comparison is an
      expression, and it has a type\\[0.25em]
      \textbf{(5)} g) \code{=} \quad \textbf{(6)} j) \code{==} --- two equals signs\\[0.25em]
      \textbf{(7)} i) \textbf{both} \quad \textbf{(8)} k) \textbf{either} --- at least one.

\begin{frame}[fragile]{Which of these compile? \timing{Alone}{60 seconds}}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
\begin{lstlisting}[language=Java]
int i = 15;

boolean a = 10 + 3;               // 1
boolean b = i > 10 && i < 20;     // 2
int     c = 10 + true;            // 3
if (i = 15) { ... }               // 4
boolean d = !(i == 100);          // 5
\end{lstlisting}
    \vspace{0.2em}
    \begin{discussionbox}{For each line}
      \small Compiles, or not? If not --- what is the compiler complaining about?
    \end{discussionbox}
    \column{0.44\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{1} \no{} \code{int cannot be converted to boolean}. Not the number ---
      the \textbf{type of the expression}\\[0.25em]
      \textbf{2} \yes{} two \code{boolean}s joined by \code{\&\&} is a \code{boolean}\\[0.25em]
      \textbf{3} \no{} \code{boolean} is its own type, not a number. Valid C++; not
      Java\\[0.25em]
      \textbf{4} \no{} \code{=} assigns, so this is an \code{int}, and an \code{if}
      needs a \code{boolean}. The classic typo, caught\\[0.25em]
      \textbf{5} \yes{} \code{d} is \code{true}
    \end{contextbox}
  \end{columns}
  \keyline{Three errors, one idea: expressions have types.}
\end{frame}

\begin{frame}[fragile]{True or false? \timing{Alone}{60 seconds}}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
\begin{lstlisting}[language=Java]
int i = 15;

i > 10 && i < 20          // (a)
i < 10 || i > 20          // (b)
!(i == 15)                // (c)
i != 15 || i >= 15        // (d)
i > 15                    // (e)
\end{lstlisting}
    \vspace{0.2em}
    {\small And: how would you write (a) in Python?}
    \column{0.46\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{(a)} \code{true} --- both sides are true\\[0.25em]
      \textbf{(b)} \code{false} --- \emph{neither} side is\\[0.25em]
      \textbf{(c)} \code{false} --- \code{i == 15} is true, and \code{!} flips it\\[0.25em]
      \textbf{(d)} \code{true} --- the left side is false, but \code{||} only needs
      one\\[0.25em]
      \textbf{(e)} \textcolor{badred}{\textbf{\code{false}}} --- 15 is not greater than 15.
      \code{i >= 15} would be true\\[0.35em]
      Python: \code{i > 10 and i < 20}. The words are spelled out
    \end{contextbox}
  \end{columns}
  \keyline{> and >= differ on exactly one value. That value is where the bugs live.}
\end{frame}

\begin{frame}[fragile]{Two \texttt{if}s, again \timing{Alone}{60 seconds}}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
\begin{lstlisting}[language=Java]
if (mark < 50) {
    System.out.println("Fail");

}
if (mark >= 50) {
    System.out.println("Pass");
}
\end{lstlisting}

    \begin{discussionbox}{What prints when \code{mark} is\ldots}
      \small
      \textbf{A.} 45 \qquad \textbf{B.} 38 \qquad \textbf{C.} 50\\[0.2em]
      \textbf{D.} One word fixes it. Which, and where?
    \end{discussionbox}
    \column{0.44\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{A.} \code{Fail} \emph{and} \code{Pass}. The first block made it 55, and the
      second \code{if} tests the \textbf{new} value\\[0.25em]
      \textbf{B.} \code{Fail} only --- 48 is still below 50\\[0.25em]
      \textbf{C.} \code{Pass} only\\[0.25em]
      \textbf{D.} \code{else} --- in place of the second \code{if (mark >= 50)}. Now
      \textbf{exactly one} runs, whatever the first block does to \code{mark}
    \end{contextbox}
  \end{columns}
  \keyline{Two ifs give you no guarantee. An else guarantees exactly one.}
\end{frame}

\begin{frame}[fragile]{Order decides the winner}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
\begin{lstlisting}[language=Java]
int kg = 25;
if (kg > 0) {
    System.out.println("Small: $5");
} else if (kg > 10) {
    System.out.println("Medium: $12");
} else if (kg > 20) {
    System.out.println("Large: $20");
} else {
    System.out.println("Invalid weight");
}
\end{lstlisting}
    \begin{discussionbox}{Three questions}
      \small
      \textbf{1.} What prints?\quad \textbf{2.} Which branches can \emph{never} run?\quad
      \textbf{3.} Any warning?
    \end{discussionbox}
    \column{0.46\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{1.} \code{Small: \$5} --- for a 25\,kg parcel\\[0.3em]
      \textbf{2.} \code{Medium} and \code{Large}. Anything above 10 is also above 0, and the
      ladder stops at the \textbf{first} true test\\[0.3em]
      \textbf{3.} No. It compiles, runs, and quietly undercharges\\[0.4em]
      \goodlabel{Fix} test \code{kg > 20} first, then \code{kg > 10}, then \code{kg > 0}:
      \textbf{specific before general}
    \end{contextbox}
  \end{columns}
  \keyline{The compiler checks your types. It does not check your logic.}
\end{frame}

if (temp < 20) {

\section{\texttt{switch} and fall-through}
% ======================================================================

\begin{frame}[fragile]{Predict the output \timing{Alone}{60 seconds}}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
\begin{lstlisting}[language=Java]
switch (day) {
    case 6 : System.out.println("Saturday");
    case 7 : System.out.println("Sunday");
             break;
    case 1 : System.out.println("Monday");
    default : System.out.println("Weekday");
}
\end{lstlisting}
    \vspace{0.2em}
    \begin{discussionbox}{What prints when \code{day} is\ldots}
      \small \textbf{A.} 6 \qquad \textbf{B.} 7 \qquad \textbf{C.} 1 \qquad \textbf{D.} 3
    \end{discussionbox}
    \column{0.44\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{A.} \code{Saturday}, \code{Sunday} --- no \code{break} after case 6, so it
      \textbf{falls through} into case 7, then stops at its \code{break}\\[0.25em]
      \textbf{B.} \code{Sunday} --- it starts \emph{at the match}, not at the top\\[0.25em]
      \textbf{C.} \code{Monday}, \code{Weekday} --- falls through into
      \code{default}\\[0.25em]
      \textbf{D.} \code{Weekday} --- nothing matches, so \code{default} runs
    \end{contextbox}
  \end{columns}
  \keyline{A switch starts at the match and stops at a break --- not at the next case.}
\end{frame}

\begin{frame}[fragile]{Could this be a \texttt{switch}?}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
    \begin{discussionbox}{Yes or no --- and why?}
      \small
      \begin{enumerate}[itemsep=0.35em, leftmargin=1.3em]
        \item \code{x == 1 || x == 5 || x == 87}\\ where \code{x} is an \code{int}
        \item \code{mark >= 50}
        \item \code{c == 'y' || c == 'Y'}\\ where \code{c} is a \code{char}
        \item \code{temp == 36.6}\\ where \code{temp} is a \code{double}
        \item \code{choice} is 1, 2 or 3 --- anything else is an error
      \end{enumerate}
    \end{discussionbox}
    \column{0.49\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{1.} \yes{} equality on an \code{int}: stack \code{case 1 :}
      \code{case 5 :} \code{case 87 :} over one body\\[0.25em]
      \textbf{2.} \no{} not an equality check --- no ranges\\[0.25em]
      \textbf{3.} \yes{} \code{char} is switchable: \code{case 'y' : case 'Y' :}\\[0.25em]
      \textbf{4.} \no{} \code{double} is not switchable. Only \code{byte}, \code{short},
      \code{char}, \code{int}, \code{String} (and enums, later)\\[0.25em]
      \textbf{5.} \yes{} three cases; the error goes in \code{default}
    \end{contextbox}
  \end{columns}
  \keyline{switch is a restricted tool: equality only, on a short list of types.}
\end{frame}

% ======================================================================

\begin{frame}[fragile]{Spot the errors: Python \timing{Alone}{60 seconds}}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \badlabel{Five things are wrong. Someone learnt Java first.}
\begin{lstlisting}[style=bad, language=Py, numbers=left, numberstyle=\tiny\color{codegray}, numbersep=0.9em]
x = int(input("Enter a number: "))
if (x > 0 && x < 10) {
    print("small")
else if x == 10:
print("ten")
else
    print("big")
\end{lstlisting}
    \column{0.46\textwidth}
    \pause
    \goodlabel{Fixed}
\begin{lstlisting}[style=good, language=Py]
x = int(input("Enter a number: "))
if x > 0 and x < 10:
    print("small")
elif x == 10:
    print("ten")
else:
    print("big")
\end{lstlisting}
    \begin{itemize}[itemsep=0.15em]
      \footnotesize
      \item Line 2: \code{and}, not \code{\&\&}; a \textbf{colon}, not \code{\{}
      \item Line 4: \code{elif}, not \code{else if}
      \item Line 5: the body must be \textbf{indented} --- indentation \emph{is} the block
      \item Line 6: \code{else} needs its colon
    \end{itemize}
  \end{columns}
  \keyline{The brackets around the condition are allowed. They are just not Python style.}
\end{frame}

\begin{frame}[fragile]{\texttt{match}: predict, then compare}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
\begin{lstlisting}[language=Py]
match day:
    case 6 | 7:
        print("Weekend")
    case 1:
        print("Monday")
    case _:
        print("Weekday")
\end{lstlisting}
    \vspace{0.2em}
    \begin{discussionbox}{Three questions}
      \small
      \textbf{1.} \code{day = 1}: what prints? (Java's \code{switch} printed two
      lines for 1.)\\[0.15em]
      \textbf{2.} \code{case 6 | 7} and \code{case \_}: what are they in Java?\\[0.15em]
      \textbf{3.} Delete \code{case \_}. What prints for \code{day = 9}?
    \end{discussionbox}
    \column{0.49\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{1.} \code{Monday} only. \textbf{No fall-through}: only the matching block
      runs, so there is no \code{break} to forget\\[0.25em]
      \textbf{2.} Stacked labels \code{case 6 : case 7 :}, and \code{default :}\\[0.25em]
      \textbf{3.} \textbf{Nothing}, and no error. Handle the case you are sure cannot
      happen --- it costs two lines
    \end{contextbox}
    \vspace{0.2em}
    {\scriptsize
    \begin{tabular}{@{}lll@{}}
      \toprule
       & \textbf{Java \code{switch}} & \textbf{Python \code{match}}\\
      \midrule
      Fall-through & yes --- needs \code{break} & no --- never\\
      Grouping & stacked \code{case} labels & \code{case a | b:}\\
      Default & \code{default :} & \code{case \_:}\\
      \bottomrule
    \end{tabular}}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{One answer --- and three ways to get it wrong}
  \begin{columns}[T,onlytextwidth]

    \goodlabel{One way}
\begin{lstlisting}[style=good, language=Java]
if (age < 0 || age > 120) {
    System.out.println("Invalid age");
} else if (age < 18) {
    System.out.println("Child: $8");
} else if (age >= 65) {
    System.out.println("Senior: $10");
} else {
    System.out.println("Adult: $15");

\end{lstlisting}
    \pause
    \begin{discussionbox}{Each of these compiles. What goes wrong?}
      \small
      \textbf{A.} \code{age < 0 \&\& age > 120}\\[0.1em]
      \textbf{B.} \code{age > 65} instead of \code{age >= 65}\\[0.1em]
      \textbf{C.} test \code{age < 18} before the invalid test
    \end{discussionbox}
    \column{0.46\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{A.} Never true: no age is below 0 \emph{and} above 120. -1 prints
      \code{Child}, 130 prints \code{Senior}\\[0.2em]
      \textbf{B.} A 65-year-old pays \code{Adult}. Off by one\\[0.2em]
      \textbf{C.} -1 is under 18 too: \code{Child: \$8}\\[0.3em]
      \textbf{All three} compile and run. Only the boundary inputs catch them ---
      which is why the six test inputs were 17/18, 64/65, and two that ``can never
      happen''.
    \end{contextbox}
  \end{columns}
\end{frame}
