\begin{frame}[fragile]{Week 12 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{1}] \textbf{Trace with meaning.} Say what a variable \emph{means}
            after each step: \code{passed} = passes seen so far. A mark of exactly 50
            tells \code{>=} from \code{>}.
      \item[\textbf{2}] \textbf{Mutation vs rebinding.} \code{list.add(x)} and
            \code{b.value = 7} change the \emph{shared} object: the caller sees it.
            \code{b = new Box(9)} only moves the \emph{local} name.
      \item[\textbf{3}] \textbf{A pipeline of stages.} Read, parse, validate, store,
            report: one job each. A number is not yet a \emph{valid} mark.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{4}] \textbf{An error policy.} Say what happens to a bad row
            (report and skip, or stop). A rejected row never enters the accepted
            list; count from that list.
      \item[\textbf{5}] \textbf{Tests that can fail.} Empty, one item, the exact
            boundary, bad text, a blank line. A good test \emph{fails} when the rule
            is broken.
    \end{itemize}
\begin{lstlisting}
read      "Mai,50"
parse     name "Mai", mark 50
validate  is 0 <= 50 <= 100?  no: report, skip
store     records.add(...)
report    count,total,passed
\end{lstlisting}

\begin{frame}<1-2>[fragile]{\qb{2}{Multiple choice}Which test reveals the bug?}
  \neutrallabel{Rule:} {\small \code{countPassed} returns how many marks are
  $\ge 50$. This version has \textbf{one} bug. Which test \textbf{fails} on it, and
  so reveals the bug?}\\[0.1em]
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
\begin{lstlisting}[language=Java]
static int countPassed(int[] marks) {
    int count = 0;
    for (int i = 0; i < marks.length; i++)
        if (marks[i] > 50) count++;
    return count;
}
\end{lstlisting}
    \begin{discussionbox}{Choose one (90 seconds)}
      \small
      \wrongopt{A}{marks \code{\{\}}, expect 0}\\[0.15em]
      \wrongopt{B}{marks \code{\{49, 85\}}, expect 1}\\[0.15em]
      \rightopt{C}{marks \code{\{50, 70\}}, expect 2}\\[0.15em]
      \wrongopt{D}{marks \code{\{0, 100\}}, expect 1}
    \end{discussionbox}
    \column{0.50\textwidth}
    \alt<1>{%
    {\small \neutrallabel{Hint:} all four expected answers are right for the
    \emph{rule}. Run each test through the code above. Which one gets a different
    number?}}{%
    \begin{contextbox}{Answer: C}
      \footnotesize\raggedright
      \textbf{C} \yes{} the code gives \textbf{1}, not 2: \code{50 > 50} is false, so
      the mark 50 is not counted. The bug: \code{>} should be \code{>=}.\\[0.3em]
      \textbf{A, B, D} \no{} give 0, 1, 1: the right answers, \emph{even with the
      bug}. None of them has a mark of exactly 50, and for every other mark
      \code{>} and \code{>=} agree. A test that passes with the bug is weak
      evidence.\\[0.3em]
      Test \textbf{at} the boundary, not just near it.
    \end{contextbox}}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{4}{Predict the output}A box in a list}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
\begin{lstlisting}[language=Java]
class Box {
    int value;
    Box(int v) { value = v; }
}
// change and main are in class Demo
static void change(Box b, ArrayList<Box> list) {
    b.value = 7;
    b = new Box(9);
    list.add(b);
}
// in main
Box x = new Box(1);
ArrayList<Box> boxes = new ArrayList<>();
boxes.add(x);
change(x, boxes);
System.out.println(x.value + " " + boxes.size());
System.out.println(boxes.get(1).value);

System.out.println(x.value);
\end{lstlisting}
    \column{0.41\textwidth}
    \begin{onlyenv}<1>
    \begin{discussionbox}{Predict (90 seconds, alone)}
      \small\raggedright
      \textbf{1.} Write down the \textbf{three} lines that are printed.\\[0.2em]
      \textbf{2.} Which line of \code{change} affects the caller's box, and which
      line affects only \code{change}?
    \end{discussionbox}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{1.} \code{7 2}\\
      \phantom{\textbf{1.}} \code{9}\\
      \phantom{\textbf{1.}} \code{8}\\[0.2em]
      \code{b} starts as a copy of the reference in \code{x}: one box, two names.
      The list holds that box at index 0 and the new box (9) at index 1.

      \textbf{2.} \code{b.value = 7} \textbf{mutates} the shared box, so \code{x}

\begin{frame}<1-2>[fragile]{\qb{5}{Predict the output}Items on sale}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
\begin{lstlisting}[language=Py,breaklines=false]
class Item:
    def __init__(self, name, price):
        self.name, self.price = name, price
    def cost(self, n):
        if n <= 0:
            raise ValueError("bad n")
        return self.price * n
class Sale(Item):
    def cost(self, n):
        return super().cost(n) - 1
orders = [(Item("pen", 2), 3),
          (Sale("ink", 5), 2),
          (Item("pad", 4), 0)]
total = 0
for item, n in orders:
    try:
        total += item.cost(n)
        print(item.name, total)
    except ValueError:
        print("skip", item.name)
\end{lstlisting}
    \column{0.46\textwidth}
    \begin{onlyenv}<1>
    \begin{discussionbox}{Predict (90 seconds, alone)}
      \small\raggedright
      \textbf{1.} Write down the \textbf{three} lines that are printed.\\[0.2em]
      \textbf{2.} If the ink order had \code{n = 0} instead of 2, what would its line
      be? Why?
    \end{discussionbox}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{1.} \code{pen 6}\\
      \phantom{\textbf{1.}} \code{ink 15}\\
      \phantom{\textbf{1.}} \code{skip pad}\\[0.2em]
      \code{item.cost(n)} runs the method of the object's \textbf{own} class. For
      the \code{Sale}, that is $5 \times 2 - 1 = 9$, so the total is $6 + 9 = 15$.
      For \code{pad}, \code{n} is 0: \code{cost} raises \emph{before}
      \code{total +=} happens, so the total stays 15.\\[0.2em]
      \textbf{2.} \code{skip ink}. \code{super().cost(0)} raises, so \code{- 1}
      never runs. The exception passes up through \code{Sale.cost} to the
      \code{except}.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}The gradebook (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,breaklines=false,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
public class Gradebook {
    private int count, total, passed;
    public boolean add(int mark) {
        count++;
        if (mark < 0 || mark > 100) return false;
        total += mark;
        if (mark > 50) passed++;
        return true;
    }
    public int getCount() { return count; }
    public int getPassed() { return passed; }
    public double average() {
        double avg;
        if (count > 0) avg = total / count;
        return avg;
    }
}
\end{lstlisting}
    {\small \badlabel{Find 4 bugs} (3 min): 1 compile error, 3 logic errors. Give each
    line number and its fix.}
    \column{0.44\textwidth}
    \begin{onlyenv}<1>
    \neutrallabel{What the class should do}
    \begin{itemize}[itemsep=0.05em, topsep=0.1em, leftmargin=1em]
      \footnotesize
      \item \code{add}: outside 0..100 $\to$ \code{false}, no change
      \item \code{getCount}: accepted; \code{getPassed}: $\ge 50$
      \item \code{average()}: exact average of accepted marks; 0.0 if none
    \end{itemize}
    \neutrallabel{Test} {\footnotesize (\code{main} in class \code{Demo})}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
Gradebook g = new Gradebook();
System.out.println(g.add(50) + " "
    + g.add(-5) + " " + g.add(75));
System.out.println(g.getCount() + " "
    + g.getPassed());
System.out.println(g.average());
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
true false true
2 2
62.5
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.25em, leftmargin=1em]
      \footnotesize
      \item \textbf{Line 13} (compile error, reported at line 15:
            \code{variable avg might not have been initialized}): if
            \code{count} is 0, \code{avg} has no value. Use \code{double avg = 0;}
      \item \textbf{Line 4}: counts \emph{before} it checks, so $-5$ counts. Move
            \code{count++} below line 5.
      \item \textbf{Line 7}: \code{> 50} misses exactly 50. Use \code{>= 50}.
      \item \textbf{Line 14}: \code{int / int} drops the \code{.5}: $125 / 2$ is
            62. Use \code{(double) total / count}.
    \end{itemize}
    {\footnotesize Fix only line 13 and the test prints \code{3 1} and
    \code{41.0}.}
    \end{onlyenv}
  \end{columns}
\end{frame}

static Student parse(String row) {

def summarise(path):
    count, passed = 0, 0
    with open(path) as f:
        for line in f:
            line = line.strip()
            if line == "":
                continue
            try:
                mark = int(line.split(",")[1])
            except ValueError:
                print("Invalid:", line)
                continue
            count += 1
            if mark >= 50:
                passed += 1
    return count, passed

class InRange implements Check {
    private final int lo, hi;
    InRange(int lo, int hi) {
        this.lo = lo; this.hi = hi;
    }
    public boolean ok(int mark) {
        return lo <= mark && mark <= hi;
    }
}
static ArrayList<Integer> keep(
        ArrayList<Integer> marks, Check c) {
    ArrayList<Integer> result = new ArrayList<>();
    for (int m : marks)
        if (c.ok(m)) result.add(m);
    return result;
}
