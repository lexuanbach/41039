\begin{frame}[fragile]{Week 9 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{1}] \textbf{An exception jumps.} A failing statement skips the
            rest of the \code{try}; control goes to the first matching \code{catch} /
            \code{except}. No match: the program stops with a \emph{stack trace}.
      \item[\textbf{2}] \textbf{Catch what you expect.} \code{Integer.parseInt}
            throws \code{NumberFormatException}; Python's \code{int()} raises
            \code{ValueError}. Python's \code{else:} runs only if nothing failed.
      \item[\textbf{3}] \textbf{Checked or unchecked.} \code{IOException} is
            \emph{checked}: catch it or declare \code{throws}. Others, like
            \code{NumberFormatException}, are unchecked.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{4}] \textbf{Throw and clean up.} \code{throw} / \code{raise}
            reports a broken rule to the caller. \code{finally} runs on success
            \textbf{and} on failure. try-with-resources and \code{with} close a
            reader for you.
      \item[\textbf{5}] \textbf{Know the reading contract.} \code{Scanner} reads
            tokens, skipping spaces and newlines. \code{readLine()}: \code{""} is a
            blank line, \code{null} is the end. Python \code{readline()}:
            \code{"\textbackslash n"} is a blank line, \code{""} is the end.
    \end{itemize}
\begin{lstlisting}[language=Java]
String line;
while ((line = in.readLine()) != null)
    System.out.println(line);  // "" if blank
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{4}{Predict the output}Marks through method calls}
  \begin{columns}[T,onlytextwidth]

\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.4}{7.3}\selectfont]
static int mark(String s) {
    int m = Integer.parseInt(s);
    if (m > 100)
        throw new IllegalArgumentException("too big");
    return m;
}
static int total(String[] marks) {
    int sum = 0;
    for (String s : marks) {
        try {
            sum += mark(s);
        } catch (NumberFormatException e) {
            System.out.println("skip " + s);
        } finally {
            System.out.println("checked " + s);

}

}

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}The mark reader (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
import java.io.*;
public class MarkReader {
    static int sumMarks(BufferedReader in) {
        int total = 0, n = 0;
        String line;
        try {
            while ((line = in.readLine()) != null) {
                if (line.isEmpty()) break;
                n++;
                total += Integer.parseInt(line);
            }
        } catch (NumberFormatException e) {
            System.out.println("bad line " + n);
        }
        return total;
    }
}
\end{lstlisting}
    {\small \badlabel{Find 4 bugs} (3 min): 1 compile error, 3 logic errors. Give each
    line number and its fix.}
    \column{0.44\textwidth}
    \begin{onlyenv}<1>
    \neutrallabel{What the method should do}
    \begin{itemize}[itemsep=0.05em, topsep=0.1em]
      \footnotesize
      \item One mark per line; read to the end
      \item Skip blank lines, but count them (from 1)
      \item Not a number: print \code{bad line N}, \textbf{go on}
      \item Return the total of the good marks
    \end{itemize}
    \neutrallabel{Test} {\footnotesize (\code{main} in \code{Demo}, with
    \code{throws IOException})}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
String text = "50\n\nabc\n30\nxyz\n20";
BufferedReader in = new BufferedReader(
    new StringReader(text));
System.out.println(MarkReader.sumMarks(in));
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
bad line 3
bad line 5
100
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.2em, leftmargin=1em]
      \footnotesize
      \item \textbf{Line 3} (compile error, reported at line 7:
            {\scriptsize\code{unreported exception IOException; must be caught or
            declared to be thrown}}): add \code{throws IOException}.
      \item \textbf{Line 8}: \code{break} stops at the blank line. Use
            \code{continue}.
      \item \textbf{Line 9}: blank lines are not counted. Move \code{n++} above
            line 8.
      \item \textbf{Lines 6, 12}: the \code{try} wraps the whole loop, so the first
            bad line ends the reading. Put \code{try}/\code{catch} around line 10
            only.
    \end{itemize}
    {\footnotesize Fix only line 3 and it prints just \code{50}.}

class Marks {
    static int parseMark(String text) {
        String t = text.trim();
        int m;
        try {
