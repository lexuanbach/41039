\begin{frame}[fragile]{Week 6 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{1}] \textbf{Contract.} An \emph{invariant} is a rule that is true
            for every object, e.g.\ $0 \le \code{fuel} \le \code{capacity}$. Each
            method \textbf{checks before it changes}, so a refused call returns
            \code{false} and changes nothing.
      \item[\textbf{2}] \textbf{\code{private} vs \code{\_}.} Java \code{private}: the
            compiler refuses access from outside the class. Python \code{self.\_x}:
            only a \emph{convention} (``please do not touch'').
      \item[\textbf{3}] \textbf{References.} \code{new} makes an object.
            \code{b = a} copies the \emph{reference}, not the object: both names now
            reach the same object.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{4}] \textbf{Shared vs per-object data.}
            Java \code{static} field / Python class attribute: \textbf{one} copy for
            the whole class. Instance field / \code{self.x = ...} in
            \code{\_\_init\_\_}: one copy \textbf{per object}.
      \item[\textbf{5}] \textbf{Composition.} An object can \emph{have} another object
            as a field: a \code{Shelf} has a \code{Book}. Two holders may share one
            object.
    \end{itemize}
\begin{lstlisting}[language=Java]
class Ticket {
    static int issued;  // one, shared
    final int number;   // one per ticket
    private Seat seat;  // has a Seat
}
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{2}{Multiple choice}Which \code{take(n)} keeps the contract?}
  \neutrallabel{Contract:} {\small accept only $0 < n \le$ \code{stock}; a refused
  request returns \code{false} and changes \textbf{nothing}. \code{stock} starts at
  4.}\\[0.1em]
  \begin{minipage}[t]{0.315\textwidth}
    \neutrallabel{Version 1}
\begin{lstlisting}[language=Java]
boolean take(int n) {
    if (n <= 0 || n > stock)
        return false;
    stock -= n;
    return true;
}
\end{lstlisting}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.315\textwidth}
    \neutrallabel{Version 2}
\begin{lstlisting}[language=Java]
boolean take(int n) {
    stock -= n;
    if (stock < 0)
        return false;
    return true;
}
\end{lstlisting}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.315\textwidth}
    \neutrallabel{Version 3}
\begin{lstlisting}[language=Java]
boolean take(int n) {
    if (n > stock)
        return false;
    stock -= n;
    return true;
}
\end{lstlisting}
  \end{minipage}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \column{0.36\textwidth}
    \begin{discussionbox}{Choose one (90 seconds)}
      \small
      \rightopt{A}{Version 1 only}\\[0.15em]
      \wrongopt{B}{Versions 1 and 3}\\[0.15em]
      \wrongopt{C}{Versions 2 and 3}\\[0.15em]
      \wrongopt{D}{All three}
    \end{discussionbox}
    \column{0.60\textwidth}
    \alt<1>{%
    {\small \neutrallabel{Hint:} all three compile. Try \code{take(9)} (too many),
    then \code{take(-5)} (negative) on each version, and watch \code{stock}.}}{%
    \begin{contextbox}{Answer: A}
      \footnotesize\raggedright
      \textbf{V1} \yes{} \code{take(9)}, \code{take(-5)}: both \code{false}, stock
      stays 4.\\[0.1em]
      \textbf{V2} \no{} changes \emph{before} it checks: \code{take(9)} says
      \code{false} but stock is now $-5$.\\[0.1em]
      \textbf{V3} \no{} forgets $n > 0$: \code{take(-5)} says \code{true} and stock
      \emph{grows} to 9.
    \end{contextbox}}
  \end{columns}
\end{frame}

    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{1.} \code{1 2}\\
      \phantom{\textbf{1.}} \code{11 2}\\

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}The car park (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
public class CarPark {
    private int capacity;
    private static int parked;
    public CarPark(int cap) { capacity = cap; }
    public boolean park() {
        if (parked > capacity) return false;
        parked++;
        return true;
    }
    public boolean leave() {
        parked--;
        return true;
    }
    public static int free() {
        return capacity - parked;
    }
}
\end{lstlisting}
    {\small \badlabel{Find 4 bugs} (3 min): 1 compile error, 3 logic errors. Give each
    line number and its fix.}
    \column{0.44\textwidth}
    \begin{onlyenv}<1>
    \neutrallabel{What the class should do}
    \begin{itemize}[itemsep=0.05em, topsep=0.1em]
      \footnotesize
      \item \code{park()} returns \code{false} when full
      \item \code{leave()} returns \code{false} when empty
      \item \code{free()} gives the number of free spaces
      \item Two car parks never share their counts
    \end{itemize}
    \neutrallabel{Test}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
CarPark a = new CarPark(2);
CarPark b = new CarPark(5);
System.out.println(a.park() + " "
    + a.park() + " " + a.park());
System.out.println(b.free());
System.out.println(b.leave() + " "
    + b.leave() + " " + b.free());
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
true true false
5
false false 5
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.25em, leftmargin=1em]
      \footnotesize
      \item \textbf{Line 14} (compile error, reported at line 15): a \code{static}
            method has no object, so it cannot read \code{capacity}. Remove
            \code{static}.
      \item \textbf{Line 3}: \code{static} gives \textbf{all} car parks one shared
            count. Remove \code{static}.
      \item \textbf{Line 6}: \code{>} lets a 3rd car into 2 spaces. Use \code{>=}.
      \item \textbf{Line 11}: no check, so \code{parked} can go negative. Add
            \code{if (parked == 0) return false;} first.
    \end{itemize}
    {\footnotesize Fix only line 14 and \code{a} still takes 3 cars.}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{7}{Find the bug}The team (Python)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Py,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
class Team:
    members = []
    def __init__(self, name, size):
        self.name = name
        self.size = size
    def join(self, person):
        self.members.append(person)
        if len(self.members) > self.size:
            return False

class BoundedCounter {
    private final int max;
    private int value = 0;
    BoundedCounter(int max) {
        this.max = max;
    }
    boolean increment() {
        if (value >= max) return false;
        value++;
        return true;
    }

    int getValue() { return value; }
}

class Playlist:
    def __init__(self):
        self._songs = []
    def add(self, song):
        if song == "" or song in self._songs:
            return False
        self._songs.append(song)
        return True
    def __str__(self):
        text = str(len(self._songs)) + " songs"
        if self._songs:
            text += ": " + ", ".join(self._songs)
        return text
