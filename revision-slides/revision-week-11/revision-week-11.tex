\begin{frame}[fragile]{Week 11 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{1}] \textbf{One responsibility.} One clear job, so one reason to
            change. Keep \textbf{calculating} apart from \textbf{printing}:
            \code{total()} returns a number; other code shows it.
      \item[\textbf{2}] \textbf{Information Expert.} Put a calculation in the class that
            \emph{has} the data. \code{Order} \textbf{asks} each \code{Line} for its
            subtotal; it never reaches into its \code{private} fields.
      \item[\textbf{3}] \textbf{Depend on an interface, and be given it.} \code{Basket}
            keeps a \code{PriceRule} passed to its constructor (Python: pass a
            function). A new rule is a new class; \code{Basket} does not change.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{4}] \textbf{Keep the promise.} A replacement must \emph{behave} as
            callers expect, not just share method names. Ask only for what you use
            (e.g.\ just \code{send}).
      \item[\textbf{5}] \textbf{Cohesion and coupling.} High cohesion: parts serve one
            purpose. Low coupling: few details of others. \textbf{DRY}: one rule in
            one place. \textbf{YAGNI}: no code for imagined futures.
    \end{itemize}
\begin{lstlisting}[language=Java]
int total() {    // in class Order
    int sum = 0;
    for (Line l : lines)
        sum += l.subtotal();  // ask
    return sum;               // no print
}
\end{lstlisting}

interface Tally {
    // Promise: total() is the sum of
    // every n given to add(n).

\begin{frame}<1-2>[fragile]{\qb{4}{Predict the output}Library fines}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
interface Fee { int charge(int days); }
class Flat implements Fee {
    public int charge(int d) { return 5; }
}
class Daily implements Fee {
    public int charge(int d) { return 2 * d; }
}
class Library {
    private final Fee fee;
    Library(Fee fee) { this.fee = fee; }
    int fine(int daysLate) {
        if (daysLate <= 0) return 0;
        return fee.charge(daysLate);
    }
}
// in main, in another class Demo
Library a = new Library(new Daily());
Library b = new Library(new Flat());
System.out.println(a.fine(3) + " " + b.fine(3));
System.out.println(a.fine(0) + " " + b.fine(9));
\end{lstlisting}
    \column{0.46\textwidth}
    \begin{onlyenv}<1>
    \begin{discussionbox}{Predict (90 seconds, alone)}
      \small\raggedright
      \textbf{1.} Write down the \textbf{two} lines that are printed.\\[0.2em]
      \textbf{2.} The library wants a new fee: 1 per day, at most 4. Which existing
      class must you edit?
    \end{discussionbox}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{1.} \code{6 5}\\
      \phantom{\textbf{1.}} \code{0 5}\\[0.2em]
      Each library uses the fee it was \textbf{given}: \code{Daily} gives
      $2 \times 3 = 6$; \code{Flat} is always 5. For 0 days \code{Library} returns 0
      \emph{before} it asks the fee.\\[0.2em]
      \textbf{2.} None. Write a new class\\
      {\scriptsize\code{class Capped implements Fee \{}\\
      \code{~~public int charge(int d)}\\
      \code{~~~~\{ return Math.min(d, 4); \} \}}}\\
      and pass \code{new Capped()}. \code{fine(3)}, \code{fine(9)}, \code{fine(0)} give
      \code{3 4 0}. The ``0 if not late'' rule stays in \code{Library}.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{5}{Predict the output}A screen and a recorder}

\begin{lstlisting}[language=Py]
class Screen:
    def send(self, text):
        print("screen:", text)

class Recorder:
    def __init__(self):
        self.sent = []
    def send(self, text):
        self.sent.append(text)

def remind(channel, names):
    for name in names:
        if name != "":
            channel.send("Due: " + name)

remind(Screen(), ["An", ""])
r = Recorder()
remind(r, ["Binh", "Chi"])
print(len(r.sent), r.sent[0])
\end{lstlisting}
    \column{0.46\textwidth}
    \begin{onlyenv}<1>
    \begin{discussionbox}{Predict (90 seconds, alone)}
      \small\raggedright
      \textbf{1.} Write down the \textbf{two} lines that are printed.\\[0.2em]
      \textbf{2.} \code{remind} only ever calls \code{send}. Why does that make
      \code{remind} easy to test?
    \end{discussionbox}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{1.} \code{screen: Due: An}\\
      \phantom{\textbf{1.}} \code{2 Due: Binh}\\[0.2em]
      The empty name is skipped. \code{Screen} prints at once. \code{Recorder}
      prints \textbf{nothing}: it keeps both messages in its list.\\[0.2em]
      \textbf{2.} \code{remind} needs \textbf{only} an object with \code{send}. A test
      passes a \code{Recorder}, then checks \code{r.sent}: nothing to read off a
      screen. Asking for one small capability keeps \code{remind} loosely
      coupled.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}The library member (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
import java.util.*;
class Loan {
    private final int due;
    Loan(int due) { this.due = due; }
    boolean isLate(int day) { return day >= due; }
}
public class Member {
    private List<Loan> loans = new ArrayList<>();
    List<Loan> getLoans() { return loans; }
    boolean borrow(Loan l) {
        if (loans.size() > 3) return false;
        loans.add(l);
        return true;
    }
    int countLate(int day) {
        int n = 0;
        for (Loan l : loans) if (day > l.due) n++;
        return n;
    }
}
\end{lstlisting}
    \column{0.44\textwidth}
    \badlabel{Find 4 bugs} {\small (3 min): 1 compile error + 3 logic errors.
    Give line and fix.}\\
    \begin{onlyenv}<1>
    \neutrallabel{What the classes should do}
    \begin{itemize}[itemsep=0em, topsep=0em, leftmargin=1em]
      \footnotesize
      \item A loan is late only \textbf{after} its due day
      \item Max \textbf{3} loans; \code{getLoans()} gives a \textbf{copy}
    \end{itemize}
    \neutrallabel{Test} {\footnotesize (\code{main} in another class)}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
Member m = new Member();
m.borrow(new Loan(10));
m.borrow(new Loan(12));
System.out.println(m.countLate(10));
m.borrow(new Loan(20));
System.out.println(m.borrow(new Loan(20)));
m.getLoans().clear();
System.out.println(m.getLoans().size());
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
0
false
3
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.25em, leftmargin=1em]
      \footnotesize
      \item \textbf{Line 17} (compile error: \code{due has private access in Loan}):
            \code{Member} reaches into \code{Loan}. Ask the expert:
            \code{if (l.isLate(day)) n++;}
      \item \textbf{Line 5}: \code{>=} makes a loan late \emph{on} its due day. Use
            \code{>}.
      \item \textbf{Line 11}: \code{>} lets a 4th loan in. Use \code{>=}.
      \item \textbf{Line 9}: hands out the \textbf{real} list, so outside code can
            clear it. Return a copy: \code{new ArrayList<>(loans)}.
    \end{itemize}
    {\footnotesize Fix only line 17: the test prints \code{1}, \code{true}, \code{0}.}
    \end{onlyenv}

class Gradebook:
    def __init__(self, pass_mark):
        self._pass_mark = pass_mark
        self._marks = []
    def add(self, mark):
        self._marks.append(mark)
    def passes(self, mark):
        return mark >= self._pass_mark
    def count_passes(self):
        n = 0
        for m in self._marks:
            if self.passes(m):
                n += 1
        return n
    def count_fails(self):
        return len(self._marks) - self.count_passes()

        if (count == 0) return false;
