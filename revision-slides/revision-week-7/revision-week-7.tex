\begin{frame}[fragile]{Week 7 in five ideas (read this first)}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{1}] \textbf{A list grows.} Indexes: 0 to \code{size()}$-1$.
            \code{add(i,x)} \emph{inserts} (later elements shift right);
            \code{set(i,x)} \emph{replaces}. Python: \code{insert},
            \code{xs[i] = x}.
      \item[\textbf{2}] \textbf{Index or value?} \code{remove(1)} removes
            \textbf{index} 1; \code{remove(Integer.valueOf(1))} removes the
            \textbf{value} 1. Python: \code{pop(i)}, \code{remove(x)}.
      \item[\textbf{3}] \textbf{Removing while looping.} A removal shifts later
            elements left, so a forward loop skips one. Loop backwards, use
            \code{Iterator.remove()}, or build a new list.
    \end{itemize}
    \column{0.49\textwidth}
    \begin{itemize}[itemsep=0.4em]
      \small
      \item[\textbf{4}] \textbf{Nodes and queues.} A node holds a value and a link,
            \code{next}. A queue is first in, first out: Java \code{LinkedList},
            Python \code{deque}.
      \item[\textbf{5}] \textbf{Generics.} In \code{class Box<T>}, \code{T} is the
            \emph{type parameter}; in \code{Box<String>}, \code{String} is the
            \emph{type argument}: a class, so \code{Integer}, not \code{int}.
    \end{itemize}
\begin{lstlisting}[language=Java]
List<Integer> xs = new ArrayList<Integer>();
xs.add(7);                      // boxed
xs.add(0, 4);                   // [4, 7]
xs.remove(Integer.valueOf(7));  // [4]
int first = xs.get(0);          // unboxed
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{2}{Multiple choice}Which loop removes every negative?}
  \neutrallabel{Goal:} {\small remove \textbf{every} negative number from the
  \code{List<Integer> xs}, which starts as \mbox{\code{[4, -1, -2, 5]}}.}\\[0.1em]
  \begin{minipage}[t]{0.315\textwidth}
    \neutrallabel{Version 1}
\begin{lstlisting}[language=Java]
for (int i = 0;
     i < xs.size(); i++)
    if (xs.get(i) < 0)
        xs.remove(i);
\end{lstlisting}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.315\textwidth}
    \neutrallabel{Version 2}
\begin{lstlisting}[language=Java]
for (int i = xs.size() - 1;
     i >= 0; i--)
    if (xs.get(i) < 0)
        xs.remove(i);
\end{lstlisting}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.315\textwidth}
    \neutrallabel{Version 3}
\begin{lstlisting}[language=Java]
Iterator<Integer> it
    = xs.iterator();
while (it.hasNext())
    if (it.next() < 0)
        it.remove();
\end{lstlisting}
  \end{minipage}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \column{0.36\textwidth}
    \begin{discussionbox}{Choose one (90 seconds)}
      \small
      \wrongopt{A}{Version 1 only}\\[0.15em]
      \wrongopt{B}{Version 2 only}\\[0.15em]
      \rightopt{C}{Versions 2 and 3}\\[0.15em]
      \wrongopt{D}{All three}
    \end{discussionbox}
    \column{0.60\textwidth}
    \alt<1>{%
    {\small \neutrallabel{Hint:} all three compile and run (with
    \mbox{\code{import java.util.*;}}). \code{i} is an \code{int}, so

    \mbox{\code{[4, -1, -2, 5]}}.}}{%

\begin{frame}<1-2>[fragile]{\qb{5}{Predict the output}A chain of nodes}

\begin{lstlisting}[language=Py,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
class Node:
    def __init__(self, value, link=None):
        self.value = value
        self.next = link

c = Node("C")
a = Node("A", Node("B", c))
a.next.next = Node("X", c)
a.next = a.next.next
names = []
n = a
while n is not None:
    names.append(n.value)
    n = n.next
print(names)
print(a.next.value, c.next)
\end{lstlisting}
    \column{0.46\textwidth}
    \begin{onlyenv}<1>
    \begin{discussionbox}{Predict (90 seconds, alone)}
      \small\raggedright
      \textbf{1.} Write down the \textbf{two} lines that are printed.\\[0.2em]
      \textbf{2.} After line 9, which node can no longer be reached from \code{a}?
      Did any value have to move?
    \end{discussionbox}
    {\footnotesize Tip: draw each node as a box with an arrow for \code{next}.}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \begin{contextbox}{Answers}
      \footnotesize\raggedright
      \textbf{1.} \code{['A', 'X', 'C']}\\
      \phantom{\textbf{1.}} \code{X None}\\[0.2em]
      Line 7 builds A $\to$ B $\to$ C. Line 8 changes B's link to a new node X that
      links to C: A $\to$ B $\to$ X $\to$ C. Line 9 sets A's link to B's link (X):
      A $\to$ X $\to$ C. C is the last node, so \code{c.next} is \code{None}.\\[0.2em]
      \textbf{2.} B: no link leads to it any more. No value moved; only links
      changed. That is why a linked list can insert and remove without shifting.
    \end{contextbox}
    \end{onlyenv}
  \end{columns}
\end{frame}

\begin{frame}<1-2>[fragile]{\qb{6}{Find the bug}The mark book (Java)}
  \begin{columns}[T,onlytextwidth]
    \column{0.53\textwidth}
\begin{lstlisting}[language=Java,style=bad,numbers=left,numberstyle=\tiny\color{codegray},numbersep=0.7em]
import java.util.ArrayList;
public class MarkBook {
    private ArrayList<Integer> marks
        = new ArrayList<Integer>();
    void add(int m) { marks.add(0, m); }
    void change(int i, int m) {
        marks.add(i, m);
    }
    boolean removeMark(int m) {
        return marks.remove(m);
    }
    int last() {
        return marks.get(marks.size());
    }
    public String toString() { return "" + marks; }
}
\end{lstlisting}
    {\small \badlabel{Find 4 bugs} (3 min): 1 compile error, 3 logic errors (one of
    them crashes). Give each line number and its fix.}
    \column{0.44\textwidth}
    \begin{onlyenv}<1>
    \neutrallabel{What the class should do}
    \begin{itemize}[itemsep=0.05em, topsep=0.1em, leftmargin=1em]
      \footnotesize
      \item \code{add(m)} adds \code{m} at the \textbf{end}
      \item \code{change(i, m)} replaces index \code{i}
      \item \code{removeMark(m)} removes one \textbf{value} \code{m}; if there is
            none, returns \code{false}
      \item \code{last()} gives the last mark
    \end{itemize}
    \neutrallabel{Test (in \code{main})}
\begin{lstlisting}[language=Java,basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
MarkBook b = new MarkBook();
b.add(70); b.add(40); b.add(90);
b.change(1, 45);
System.out.println(b.removeMark(90) + " "
    + b.removeMark(12));
System.out.println(b + " " + b.last());
\end{lstlisting}
    \neutrallabel{Expected output}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.8}{7.9}\selectfont]
true false
[70, 45] 45
\end{lstlisting}
    \end{onlyenv}
    \begin{onlyenv}<2>
    \goodlabel{The 4 fixes}
    \begin{itemize}[itemsep=0.25em, leftmargin=1em]
