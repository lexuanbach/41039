\begin{frame}[fragile]{Which version would you rather debug?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \badlabel{Version 1}
\begin{lstlisting}[language=Java,style=bad]
static final int HEARTS = 0, SPADES = 1,
                 DIAMONDS = 2, CLUBS = 3;
static String colour(int suit) {
    if (suit == HEARTS || suit == DIAMONDS)
        return "red";
    return "black";
}
System.out.println(colour(DIAMONDS));
System.out.println(colour(7));
\end{lstlisting}
    \column{0.48\textwidth}
    \goodlabel{Version 2}
\begin{lstlisting}[language=Java,style=good]
enum Suit {
    HEARTS, SPADES, DIAMONDS, CLUBS;
    String colour() {
        if (this == HEARTS
                || this == DIAMONDS)
            return "red";
        return "black";
    }
}
System.out.println(Suit.DIAMONDS.colour());
\end{lstlisting}
  \end{columns}
  \pause
  \begin{contextbox}{Why version 2}
    \footnotesize\raggedright
    Version 1 prints \code{red}, then \code{black} for suit 7, which does not exist, and
    nothing complains. With the enum, \code{colour(7)} does not compile (\emph{int cannot
    be converted to Suit}). Only four suits can exist, \code{==} is safe, they print as
    \code{DIAMONDS} rather than \code{2}, and the method lives with the data.
  \end{contextbox}
\end{frame}

class Stock {

    boolean take(int n) {
        if (n <= 0 || n > count) return false;
        count -= n;
        return true;
    }
}
