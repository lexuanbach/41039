\begin{frame}[fragile]{Types and names: quick quiz}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \neutrallabel{A. Which type would you declare?}
\begin{lstlisting}[language=Java]
____ studentCount = 341;
____ average      = 75.33;
____ grade        = 'D';
____ passed       = true;
\end{lstlisting}
    \vspace{0.2em}
    \neutrallabel{B. Which break Java's naming conventions?}\\[0.3em]
    {\small\ttfamily class Student \quad class student\\[0.2em]
     int MyTotal \quad int myTotal \quad Math.PI}
    \vspace{0.4em}

    \neutrallabel{C. What is the biggest number an \code{int} holds?}
    \column{0.48\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{A.} \code{int}, \code{double}, \code{char}, \code{boolean}.
      Note \code{'D'} uses single quotes --- \code{"D"} is a \code{String}.\\[0.35em]
      \textbf{B.} \badlabel{Wrong:} \code{class student} --- class names start
      \textbf{upper}case --- and \code{int MyTotal} --- variables start
      \textbf{lower}case. \code{Math.PI} is a constant, so all caps is right.
      Conventions, not rules: the compiler accepts every one of them.\\[0.3em]
      \textbf{C.} \code{2,147,483,647} --- that is $2^{31}-1$, and it is why an \code{int}
      broke YouTube's view counter.
    \end{contextbox}
  \end{columns}
  \keyline{In practice: int, double, char, boolean.}
\end{frame}

\begin{frame}[fragile]{Predict the output \timing{Alone}{60 seconds}}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
\begin{lstlisting}[language=Java]
int a = 7;
int b = 2;

System.out.println(a / b);      // (a)
System.out.println(7.0 / b);    // (b)
System.out.println(a % b);      // (c)
System.out.println("a = " + a); // (d)
System.out.println(a + b);      // (e)

a = a + 1;
System.out.println(a);          // (f)
\end{lstlisting}
    \column{0.49\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{(a)} \code{3} --- int divided by int gives an \textbf{int}. The \code{.5} is
      not rounded, it is \emph{discarded}\\[0.25em]
      \textbf{(b)} \code{3.5} --- one \code{double} operand is enough\\[0.25em]
      \textbf{(c)} \code{1} --- \code{\%} is the remainder\\[0.25em]
      \textbf{(d)} \code{a = 7} --- here \code{+} \textbf{joins Strings}\\[0.25em]
      \textbf{(e)} \code{9} --- the same \code{+}, now \textbf{adding}. What it does
      depends on the types beside it\\[0.25em]
      \textbf{(f)} \code{8} --- read \code{=} right to left: \emph{take \code{a+1},
      put it in \code{a}}
    \end{contextbox}
  \end{columns}
  \keyline{= is not equality. It is "take the right, store it in the left".}
\end{frame}

\begin{frame}[fragile]{The bug that compiles, runs, and lies}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
\begin{lstlisting}[language=Java]
int totalMarks = 226;
int studentCount = 3;

int classAverage
        = totalMarks / studentCount;

System.out.println("Average: "
        + classAverage);
\end{lstlisting}
    \vspace{0.2em}
    \begin{discussionbox}{Three questions}
      \small
      \textbf{1.} What does it print?\quad
      \textbf{2.} What \emph{should} it print?\\
      \textbf{3.} Which of the three kinds of error is this?
    \end{discussionbox}
    \column{0.44\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{1.} \code{Average: 75}\qquad \textbf{2.} \code{75.33}\\[0.3em]
      \textbf{3.} A \textcolor{badred}{\textbf{logic error}}. It compiles, it runs, and it

      \code{classAverage} a \code{double} too. Changing only the variable is not enough:
      the division has already happened.
    \end{contextbox}
  \end{columns}
  \keyline{One wrong type. No error message. This is the class of bug that makes the news.}

\section{The same ideas in Python}
% ======================================================================

\begin{frame}[fragile]{Same program, two languages: name five differences}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
\begin{lstlisting}[language=Java]
public class Greet {
    public static void main(String[] a) {
        String name = "Bach";
        int age = 20;
        boolean ok = true;
        System.out.println(name + age);
    }
}
\end{lstlisting}
    \column{0.47\textwidth}
\begin{lstlisting}[language=Python]
name = "Bach"
age = 20
ok = True
print(name + str(age))
\end{lstlisting}
    \vspace{0.3em}
    {\small Why does the Python version need \code{str(age)} when the Java one does not?}
  \end{columns}
  \pause
  \vspace{0.3em}
  \begin{contextbox}{Answers}
    \small\raggedright
    No class or \code{main}; no type on the left of \code{=}; no semicolons; no braces;
    \code{True} is capitalised where Java writes \code{true}. And the \code{str()}: Java's
    \code{+} silently converts the \code{int}; Python's \emph{refuses} to mix \code{str}
    and \code{int}, and raises a \code{TypeError}.
  \end{contextbox}
  \vspace{0.2em}
  \begin{center}\textcolor{bkblue}{\textbf{In Python variables have no type. The data still
  does. That is ``dynamically typed''.}}\end{center}
\end{frame}

\begin{frame}[fragile]{One space changes the answer}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
    \neutrallabel{A}
\begin{lstlisting}[language=Python]
a = 20
if a < 10:
    print("small")
    print("done")
\end{lstlisting}
    \neutrallabel{B}
\begin{lstlisting}[language=Python]
a = 20
if a < 10:
    print("small")
print("done")
\end{lstlisting}
    \column{0.47\textwidth}
    \begin{discussionbox}{With \code{a = 20}, what does each print?}
      \small Only the indentation of the last line differs.
    \end{discussionbox}
    \pause
    \vspace{0.25em}
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{A} prints \textbf{nothing} --- both \code{print}s are inside the \code{if}.\\[0.25em]
      \textbf{B} prints \code{done} --- that line is outside it, so it always runs.\\[0.25em]
      Four spaces, never mixed with tabs.
    \end{contextbox}
  \end{columns}
  \keyline{In Java bad indentation is ugly. In Python it is wrong.}
\end{frame}

\begin{frame}[fragile]{Read the compiler, do not argue with it}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
\begin{lstlisting}
Greeting.java:4: error: cannot find symbol
    System.out.println("Hi " + userName);
                               ^
  symbol:   variable userName
  location: class Greeting
1 error
\end{lstlisting}
    \vspace{0.2em}
    \begin{discussionbox}{Four questions}
      \small
      \textbf{1.} Which file, and which line?\quad
      \textbf{2.} Which token?\\
      \textbf{3.} What does \code{cannot find symbol} actually mean?\\
      \textbf{4.} You get 14 errors. Which do you fix?
    \end{discussionbox}
    \column{0.44\textwidth}
    \pause
    \begin{contextbox}{Answers}
      \small\raggedright
      \textbf{1.} \code{Greeting.java}, line \textbf{4}\\[0.25em]
      \textbf{2.} \code{userName} --- follow the \code{\^{}}\\[0.25em]
      \textbf{3.} \emph{You used a name I have never seen} --- a typo, a missing
      declaration, or a capitalisation slip\\[0.25em]
      \textbf{4.} The \textbf{first} one, then recompile. Errors 2--14 are usually
      downstream of error 1\\[0.3em]
      And turn on line numbers in Ed --- without them, \code{Greeting.java:4} tells you
      nothing.
    \end{contextbox}
  \end{columns}
  \keyline{Which file, which line, what kind, which token. Every time, in that order.}
\end{frame}

\begin{frame}[fragile]{In pairs: make it print \code{Bach20} \timing{Pairs}{2 minutes}}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \badlabel{This crashes. Why, and what is the smallest fix?}
\begin{lstlisting}[style=bad, language=Python]
name = "Bach"
age = 20
print(name + age)
\end{lstlisting}
    \vspace{0.2em}
    \begin{discussionbox}{Then answer this}
      \small The equivalent Java line, \code{System.out.println(name + age)},
      does \emph{not} crash. Is Java being helpful, or is it being dangerous?
    \end{discussionbox}
    \column{0.46\textwidth}
    \pause
    \goodlabel{Fix}
\begin{lstlisting}[style=good, language=Python]
print(name + str(age))
\end{lstlisting}
    \vspace{0.2em}
    {\small \code{TypeError: can only concatenate str (not "int") to str}. Python will not
    guess what you meant.}
    \vspace{0.5em}
    \begin{contextbox}{Both, honestly}
      \small\raggedright
      Java's convenience is real, and so is the risk: the day you \emph{meant} to add two
      numbers and one of them was a \code{String}, Java will happily join them and say
      nothing. Python's refusal is noisy, early and cheap.
    \end{contextbox}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{One answer --- and what was different}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
    \goodlabel{One way}
\begin{lstlisting}[style=good, language=Python]
a = int(a)
b = int(b)
total = a + b
average = total / 2
print("Sum: " + str(total))
print("Average: " + str(average))
\end{lstlisting}
    \vspace{0.2em}
    \badlabel{Skip the int() and nothing complains}
\begin{lstlisting}[style=bad, language=Python]
total = a + b      # "72" -- a logic bug
\end{lstlisting}
    \column{0.46\textwidth}
    \pause
    \begin{contextbox}{Three differences, all from Lecture 1}
      \small\raggedright
      \textbf{Input arrives as text.} Java's \code{nextInt()} converted for you;
      Python's \code{input()} does not, so you call \code{int()} --- and typing
      \code{seven} now fails \emph{there}, with a \code{ValueError}.\\[0.3em]
      \textbf{\code{/} gives 4.5 by itself.} Java's \code{9 / 2} gave 4. Same symbol,
      different bet.\\[0.3em]
      \textbf{\code{+} will not mix.} \code{"Sum: " + total} is the \code{Bach20}
      crash again, so \code{str()} goes back in.
    \end{contextbox}
  \end{columns}
\end{frame}
